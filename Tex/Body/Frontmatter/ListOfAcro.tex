\section*{Mathematical Symbols}
\begin{tabular}{lcl}
    \textbf{Symbol} &:& \textbf{Explanation}\\
    \\
    $p$ &:& Prime modulus for elliptic curve field\\
    $q$ &:& Prime modulus for Kyber polynomial ring (q = 3329)\\
    $n$ &:& Polynomial degree (n = 256 for Kyber)\\
    $G$ &:& Generator point on elliptic curve\\
    $d$ &:& Private key (scalar)\\
    $Q$ &:& Public key (point on elliptic curve)\\
    $\mathbb{Z}_q$ &:& Ring of integers modulo q\\
    $R_q$ &:& Polynomial ring $\mathbb{Z}_q[X]/(X^{256} + 1)$\\
    $\mathbf{A}$ &:& Public matrix in Kyber\\
    $\mathbf{s}$ &:& Secret vector in Kyber\\
    $\mathbf{e}$ &:& Error/noise vector in Kyber\\
    $\mathbf{t}$ &:& Public key vector in Kyber\\
    $\cdot$ &:& Scalar multiplication (elliptic curve point)\\
    $\times$ &:& Multiplication operation\\
    $\oplus$ &:& XOR (exclusive OR) operation\\
    $\pmod{m}$ &:& Modulo m operation\\
\end{tabular}
